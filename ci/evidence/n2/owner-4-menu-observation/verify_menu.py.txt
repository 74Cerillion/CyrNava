"""Owner-only N2 GNOME menu check of the already GUI-verified disposable client.

Temporarily registers the exact development Desktop Entry, observes a manual
Applications-menu launch, then removes only its unchanged registration. No install,
updater, personal profile, security change, signing or automatic GUI launch.
"""

import os
import subprocess
import sys

import traceback
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
RESULT = "cfb76d3616e0c98e307653397d45bee96b8b1f8165e4de5124d828c8ce3ecdde"
MANIFEST = "a1e066a3d769319a761b8643469f055201ef04b2b80ff92c0ac9604e1a4f8d82"


def module(name, path):
    import importlib.util

    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(result)
    return result


def register(engine, source, destination):
    """Exclusive creation; collisions and unsafe parents never get overwritten."""
    source, destination = engine.regular(source), engine.checked_path(destination)
    destination.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    info = destination.parent.stat()
    engine.require(
        info.st_uid == os.getuid() and not info.st_mode & 0o022, "Unsafe menu directory"
    )
    data = source.read_bytes()
    fd = os.open(
        destination, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o644
    )
    with os.fdopen(fd, "wb") as stream:
        stream.write(data)
        stream.flush()
        os.fsync(stream.fileno())
    engine.sync_directory(destination.parent)
    return engine.digest(destination)


def unregister(engine, destination, expected):
    """Remove only this check's exact owned registration; preserve changed files."""
    destination = engine.regular(destination)
    engine.require(
        destination.stat().st_uid == os.getuid()
        and engine.digest(destination) == expected,
        "Changed menu entry preserved; owner review required",
    )
    destination.unlink()
    engine.sync_directory(destination.parent)


def gio_discovery(destination):
    """Ask the real desktop registry by ID, not by direct filename."""
    gio = ctypes.CDLL("libgio-2.0.so.0")
    objects = ctypes.CDLL("libgobject-2.0.so.0")
    gio.g_desktop_app_info_new.argtypes = [ctypes.c_char_p]
    gio.g_desktop_app_info_new.restype = ctypes.c_void_p
    for name in (
        "g_desktop_app_info_get_filename",
        "g_app_info_get_id",
        "g_app_info_get_name",
    ):
        function = getattr(gio, name)
        function.argtypes = [ctypes.c_void_p]
        function.restype = ctypes.c_char_p
    gio.g_app_info_should_show.argtypes = [ctypes.c_void_p]
    gio.g_app_info_should_show.restype = ctypes.c_int
    objects.g_object_unref.argtypes = [ctypes.c_void_p]
    app = gio.g_desktop_app_info_new(os.fsencode(destination.name))
    if not app:
        raise ValueError("GNOME registry did not resolve the development entry")
    try:
        return {
            "correct_path": os.fsdecode(gio.g_desktop_app_info_get_filename(app))
            == str(destination),
            "correct_id": os.fsdecode(gio.g_app_info_get_id(app)) == destination.name,
            "name": os.fsdecode(gio.g_app_info_get_name(app)),
            "should_show": bool(gio.g_app_info_should_show(app)),
        }
    finally:
        objects.g_object_unref(app)


def find_gui(bootstrap):
    """Read only same-user process metadata; match the exact installed command."""
    expected = [b"-B", b"-I", os.fsencode(bootstrap)]
    matches = []
    for process in Path("/proc").iterdir():
        if not process.name.isdecimal():
            continue
        try:
            if process.stat().st_uid != os.getuid():
                continue
            command = (process / "cmdline").read_bytes().rstrip(b"\0").split(b"\0")
            if len(command) == 4 and command[1:] == expected:
                matches.append(int(process.name))
        except (OSError, ValueError):
            continue
    if len(matches) > 1:
        raise ValueError("More than one matching installed application process")
    return matches[0] if matches else None


def observe_menu_launch(check, bootstrap, *, startup_seconds=120, window_seconds=600):
    sandbox = check.support("sandbox_probe")
    deadline = time.monotonic() + startup_seconds
    while (pid := find_gui(bootstrap)) is None:
        if time.monotonic() >= deadline:
            raise TimeoutError(
                "No manual GNOME menu launch observed within two minutes"
            )
        time.sleep(0.1)
    process = Path("/proc") / str(pid)
    started = sandbox.process_start_time(process)
    owned, observed, authenticated = (
        {},
        dict.fromkeys(("backend", "renderer", "qt_providers"), False),
        False,
    )
    deadline = time.monotonic() + window_seconds
    print(
        "N2: menu-launched application observed; checking renderer and backend.",
        flush=True,
    )
    while check.still_alive(pid, started, sandbox):
        sample = check.sample_tree(pid, owned, sandbox)
        if sample.pop("foreign_qt"):
            raise RuntimeError("Foreign Qt provider rejected")
        for key, value in sample.items():
            observed[key] |= value
        if all(observed.values()) and not authenticated:
            if not check.unauthenticated_rejected():
                raise RuntimeError("Backend authentication failed")
            authenticated = True
            print(
                "N2: renderer/providers/auth PASS. Inspect existing synthetic note, image, Dark, icon and window grouping; close normally without editing.",
                flush=True,
            )
        if time.monotonic() >= deadline:
            raise TimeoutError(
                "Menu application close deadline exceeded; close it manually"
            )
        time.sleep(0.1)
    cleanup_deadline = time.monotonic() + 15
    while any(check.still_alive(p, start, sandbox) for p, start in owned.items()):
        if time.monotonic() >= cleanup_deadline:
            raise RuntimeError("Menu application's descendant cleanup incomplete")
        time.sleep(0.1)
    if not all(observed.values()) or not authenticated:
        raise RuntimeError("Menu application live evidence incomplete")
    return {
        **dict.fromkeys(observed, "PASS"),
        "unauthenticated_backend_rejected": "PASS",
        "owned_process_cleanup": "PASS",
        "exit_code": "NOT AVAILABLE for external GNOME child",
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--owner-host", action="store_true", required=True)
    args = parser.parse_args()
    del args
    engine = module("n2_menu_engine", ROOT / "desktop/install_engine.py")
    engine.host_supported()
    engine.require(
        sys.stdin.isatty()
        and os.environ.get("WAYLAND_DISPLAY")
        and os.environ.get("XDG_SESSION_TYPE") == "wayland",
        "Owner GNOME Wayland terminal required",
    )
    client = engine.checked_path(ROOT / "ci-output/native-n2-client-3")
    output = engine.checked_path(ROOT / "ci-output/native-n2-owner-4")
    engine.require(
        not output.exists(), "Previous menu attempt preserved; no automatic retry"
    )
    prior = ROOT / "ci-output/native-n2-owner-3"
    engine.require(
        engine.digest(prior / "result.json") == RESULT, "Reviewed GUI PASS changed"
    )
    approved = engine.load(prior / "result.json")
    engine.require(
        approved["status"] == "PASS"
        and approved["complete"] is True
        and approved["saved_data_and_origin"] == "PASS"
        and approved["owner_visual_confirmation"] == "PASS",
        "Completed GUI verification required",
    )
    release = (
        client
        / "releases"
        / engine.member_name(engine.load(client / "current.json")["release"])
    )
    with engine.installation_lock(client):
        receipt, manifest = engine.installed(release)
        engine.require(
            receipt["manifest_sha256"] == MANIFEST and manifest["kind"] == "candidate",
            "Wrong reviewed development package",
        )
        engine.require(
            engine.digest(ROOT / "packaging/native/verify_desktop.py")
            == approved["driver_sha256"],
            "Reviewed state verifier changed",
        )
    desktop = client / "xdg/applications" / (engine.CANDIDATE_ID + ".desktop")
    engine.require(
        desktop.read_bytes() == engine.desktop_entry(release, engine.CANDIDATE_ID, True)
        and engine.digest(desktop) == approved["desktop_sha256"],
        "Reviewed Desktop Entry changed",
    )
    data_home = engine.checked_path(
        Path(os.environ.get("XDG_DATA_HOME", Path.home() / ".local/share"))
    )
    entry = engine.checked_path(data_home / "applications" / desktop.name)
    engine.require(
        not entry.exists() and not entry.is_symlink(),
        "Existing development menu entry preserved",
    )
    engine.require(
        find_gui(client / "launch.py") is None, "Close existing candidate first"
    )
    check = module(
        "n2_menu_monitor", ROOT / "packaging/native/feasibility/check_application.py"
    )
    engine.require(
        engine.digest(ROOT / "packaging/native/feasibility/check_application.py")
        == "406e0afd6011dbf9f1e9031d8f6e8c8cdf8eb5ba3b7557b2db4fca67260584ed",
        "Accepted live policy changed",
    )
    check.CANDIDATE = release
    check.unauthenticated_rejected = lambda: unauthenticated(receipt, engine)
    output.mkdir(mode=0o700)
    (output / "active").write_text("N2 menu verification only\n")
    record = {
        "status": "FAIL",
        "complete": False,
        "scope": "N2 actual GNOME Applications-menu launch",
        "manifest_sha256": MANIFEST,
        "archive_sha256": manifest["archive_sha256"],
        "inherited_GUI_result_sha256": RESULT,
        "driver_sha256": engine.digest(Path(__file__)),
        "desktop_sha256": engine.digest(desktop),
        "checks": {},
    }
    registered = None

    def snapshot(phase):
        with (
            engine.installation_lock(client),
            (output / (phase + "-state.private.log")).open("xb") as log,
        ):
            subprocess.run(
                [
                    str(release / ".venv/bin/python"),
                    "-B",
                    "-I",
                    str(ROOT / "packaging/native/verify_desktop.py"),
                    "--owner-host",
                    "--root",
                    str(client),
                    "--output",
                    str(output),
                    "--manifest-sha256",
                    MANIFEST,
                    "--snapshot",
                    phase,
                ],
                stdout=log,
                stderr=subprocess.STDOUT,
                check=True,
                timeout=30,
            )
        return engine.load(output / (phase + "-state.json"))

    try:
        record["stage"] = "saved-state-preflight"
        before = snapshot("edit")
        engine.require(
            before == engine.load(prior / "restart-state.json"),
            "Verified saved data changed; preserve for review",
        )
        print(
            "N2: temporary 'CyrNava — N2 Candidate' GNOME menu entry; existing disposable profile only. No normal CyrNava data or security settings change.",
            flush=True,
        )
        input("Press Enter to register the development entry for this check: ")
        record["stage"] = "menu-registration"
        registered = register(engine, desktop, entry)
        discovery = gio_discovery(entry)
        engine.require(
            all(discovery[k] for k in ("correct_path", "correct_id", "should_show")),
            "GNOME discovery mismatch",
        )
        record["discovery"] = discovery
        record["checks"]["GIO_menu_discovery"] = "PASS"
        print(
            "Open GNOME Activities / Show Applications. Search 'CyrNava' and click 'CyrNava — N2 Candidate' ONCE. Do not use a terminal launch. Start within two minutes; wait for this terminal's close instruction.",
            flush=True,
        )
        record["stage"] = "owner-menu-launch"
        record["live_checks"] = observe_menu_launch(check, client / "launch.py")
        record["stage"] = "post-close-state"
        engine.require(before == snapshot("restart"), "Saved state or origin changed")
        answer = input(
            "Only if you selected the entry in GNOME Applications and the correct window/icon/grouping, saved note/image and Dark were correct, type CONFIRM N2 MENU: "
        )
        engine.require(
            " ".join(answer.split()).casefold() == "confirm n2 menu",
            "Menu visual confirmation missing",
        )
        record["checks"].update(
            owner_GNOME_menu_launch="PASS",
            owner_icon_and_window_grouping="PASS",
            exact_saved_data_and_origin="PASS",
            owner_appearance="PASS",
        )
        record.update(status="PASS", stage="complete")
    except (Exception, KeyboardInterrupt) as error:  # noqa: BLE001 - retained private diagnostic
        (output / "exception.private.log").write_text(traceback.format_exc())
        record["error_type"] = type(error).__name__
    finally:
        if registered is not None:
            try:
                unregister(engine, entry, registered)
                record["checks"]["temporary_menu_entry_removed"] = "PASS"
            except (OSError, ValueError):
                record.update(
                    status="FAIL",
                    menu_cleanup="CHANGED ENTRY PRESERVED; OWNER REVIEW REQUIRED",
                )
        (output / "active").unlink(missing_ok=True)
        record["complete"] = True
        engine.durable(output / "result.json", record)
        print(json.dumps(record, indent=2), flush=True)
    return 0 if record["status"] == "PASS" else 2


def unauthenticated(receipt, engine):
    import urllib.error
    import urllib.request

    port = engine.load(Path(receipt["profile"]) / "desktop.json")["port"]
    engine.require(type(port) is int and 0 < port < 65536, "Invalid backend port")
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    try:
        with opener.open(f"http://127.0.0.1:{port}/api/tree", timeout=3):
            return False
    except urllib.error.HTTPError as error:
        error.close()
        return error.code == 401


if __name__ == "__main__":
    os.umask(0o077)
    raise SystemExit(main())
