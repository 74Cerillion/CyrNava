/* N2 exact-path AppArmor entry. No shell, arbitrary command, setuid or fallback. */
#define _GNU_SOURCE
#include <ctype.h>
#include <fcntl.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <unistd.h>

static int refuse(const char *operation) {
    fprintf(stderr, "CyrNava native launch refused: %s\n", operation);
    return 1;
}

static int profile_valid(const char *name, uid_t uid) {
    char prefix[80];
    int n = snprintf(prefix, sizeof prefix, "cyrnava-native-u%lu-", (unsigned long)uid);
    if (n < 0 || (size_t)n >= sizeof prefix || strlen(name) != (size_t)n + 32 ||
        strncmp(name, prefix, (size_t)n) != 0) return 0;
    for (size_t i = (size_t)n; name[i]; i++)
        if (!isdigit((unsigned char)name[i]) && !(name[i] >= 'a' && name[i] <= 'f')) return 0;
    return 1;
}

static int label_matches(const char *label, const char *expected) {
    size_t n = strlen(expected);
    if (strncmp(label, expected, n)) return 0;
    const char *mode = label + n;
    return !strcmp(mode, " (enforce)\n") || !strcmp(mode, " (enforce)") ||
           !strcmp(mode, " (default_allow)\n") || !strcmp(mode, " (default_allow)") ||
           !strcmp(mode, " (allow)\n") || !strcmp(mode, " (allow)");
}

static int owned_file(const char *path, int executable) {
    struct stat info;
    return !lstat(path, &info) && S_ISREG(info.st_mode) &&
           info.st_uid == getuid() && info.st_nlink == 1 &&
           !(info.st_mode & 0077) && !(info.st_mode & 07000) &&
           (!executable || (info.st_mode & 0100));
}

static int join(char *path, size_t size, const char *root, const char *name) {
    int n = snprintf(path, size, "%s/%s", root, name);
    return n >= 0 && (size_t)n < size;
}

static int read_small(const char *path, char *buffer, size_t size) {
    int fd = open(path, O_RDONLY | O_CLOEXEC | O_NOFOLLOW);
    if (fd < 0) return 0;
    ssize_t n = read(fd, buffer, size - 1);
    char extra;
    int complete = n >= 0 && read(fd, &extra, 1) == 0;
    close(fd);
    if (!complete) return 0;
    buffer[n] = '\0';
    return !memchr(buffer, '\0', (size_t)n);
}

#ifndef CYRNAVA_UNIT_TEST
int main(int argc, char **argv) {
    if (getuid() == 0 || geteuid() != getuid()) return refuse("ordinary-user-required");
    if (!(argc == 1 || (argc == 2 && !strcmp(argv[1], "--check"))))
        return refuse("unsupported-arguments");
    const char *forbidden[] = {"LD_PRELOAD", "LD_LIBRARY_PATH", "PYTHONPATH", "PYTHONHOME",
        "QT_PLUGIN_PATH", "QML_IMPORT_PATH", "QML2_IMPORT_PATH", "QTWEBENGINE_DISABLE_SANDBOX",
        "QTWEBENGINE_CHROMIUM_FLAGS", "QTWEBENGINEPROCESS_PATH", "QTWEBENGINE_RESOURCES_PATH",
        "QTWEBENGINE_LOCALES_PATH", "QTWEBENGINE_REMOTE_DEBUGGING", "FLATPAK_ID", NULL};
    for (int i = 0; forbidden[i]; i++) {
        const char *value = getenv(forbidden[i]);
        if (value && *value) return refuse("unsafe-environment-override");
    }
    char root[PATH_MAX], path[PATH_MAX], script[PATH_MAX], expected[160], label[512];
    ssize_t length = readlink("/proc/self/exe", root, sizeof root - 1);
    if (length <= 0 || (size_t)length >= sizeof root - 1) return refuse("executable-identity");
    root[length] = '\0';
    char *name = strrchr(root, '/');
    if (!name || strcmp(name + 1, "cyrnava-native") || !owned_file(root, 1))
        return refuse("executable-identity");
    *name = '\0';
    struct stat info;
    if (lstat(root, &info) || !S_ISDIR(info.st_mode) || info.st_uid != getuid() ||
        (info.st_mode & 0077)) return refuse("private-install-root");
    if (!join(path, sizeof path, root, "apparmor-profile") || !owned_file(path, 0) ||
        !read_small(path, expected, sizeof expected)) return refuse("profile-metadata");
    size_t n = strlen(expected);
    if (n && expected[n - 1] == '\n') expected[n - 1] = '\0';
    if (!profile_valid(expected, getuid())) return refuse("profile-identity");
    if (!read_small("/proc/self/attr/apparmor/current", label, sizeof label) ||
        !label_matches(label, expected)) return refuse("apparmor-profile-not-attached");
    if (!join(script, sizeof script, root, "launch.py") || !owned_file(script, 0) ||
        !join(path, sizeof path, root, "current.json") || !owned_file(path, 0))
        return refuse("bootstrap-layout");
    char *const command[] = {"/usr/bin/python3.14", "-B", "-I", script,
                            argc == 2 ? "--check" : NULL, NULL};
    execv(command[0], command);
    return refuse("interpreter-exec");
}
#endif
