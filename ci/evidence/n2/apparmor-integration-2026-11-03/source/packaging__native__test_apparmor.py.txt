def module(name, relative):
    spec = importlib.util.spec_from_file_location(name, ROOT / relative)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


aa = module("n2_aa_tests", "packaging/native/apparmor_support.py")
adapter = module("n2_aa_gio_tests", "packaging/native/desktop_process.py")
install_tests = module(
    "n2_aa_install_fixtures", "packaging/native/test_install_engine.py"
)
engine = install_tests.engine


class PolicyTests(unittest.TestCase):
    root = Path("/home/synthetic/exact install % $")

    def test_identity_is_bound_to_exact_root_and_user(self):
        name = aa.profile_name(self.root, 1000)
        self.assertNotEqual(name, aa.profile_name(self.root / "other", 1000))
        self.assertNotEqual(name, aa.profile_name(self.root, 1001))
        self.assertRegex(name, r"^cyrnava-native-u1000-[0-9a-f]{32}$")

    def test_policy_has_exact_attachment_and_inherited_execs(self):
        release = self.root / "releases/0.1.0-n2.2-1234567890abcdef"
        policy = aa.render_policy(self.root, release, 1000).decode()
        self.assertIn("flags=(default_allow)", policy)
        self.assertIn('"/usr/bin/python3.14" ix,', policy)
        self.assertIn(aa.policy_path(self.root / "cyrnava-native"), policy)
        self.assertIn("userns create,", policy)
        self.assertIn('QtWebEngineProcess" ix,', policy)
        self.assertEqual(policy.count(" ix,"), 3)
        for forbidden in (
            "capability sys_admin",
            "**",
            "complain",
            "ux,",
            "Px,",
            "no-sandbox",
        ):
            self.assertNotIn(forbidden, policy)

    def test_pattern_or_traversal_attachment_is_rejected(self):
        for root in (
            "relative",
            "/a/../b",
            "/home/*/app",
            "/home/a[bc]",
            "/home/@{HOME}",
            "/home/a?",
            "/home/a\\b",
            "/home/a\n",
        ):
            with self.subTest(root=root), self.assertRaises(ValueError):
                aa.policy_path(root)

    def test_quotes_are_literal_and_other_release_is_rejected(self):
        self.assertEqual(aa.policy_path('/home/a"b'), '"/home/a\\"b"')
        with self.assertRaises(ValueError):
            aa.render_policy(self.root, Path("/other/releases/0.1.0-1234567890abcdef"))

    def test_label_requires_exact_identity_and_noncomplain_mode(self):
        name = aa.profile_name(self.root)
        for mode in ("enforce", "allow", "default_allow"):
            self.assertEqual(
                aa.check_label(self.root, name + f" ({mode})\n")["profile"], name
            )
        for label in (
            "unconfined\n",
            name + " (complain)",
            name + "-other (enforce)",
            "outer//&" + name + " (enforce)",
            name + " (unconfined)",
        ):
            with self.subTest(label=label), self.assertRaises(ValueError):
                aa.check_label(self.root, label)

    def test_engine_refuses_direct_python_launch_without_policy_before_reexec(self):
        parent = ROOT / "ci-output/native-n2-apparmor-tests-1"
        parent.mkdir(mode=0o700, exist_ok=True)
        with tempfile.TemporaryDirectory(dir=parent) as temporary:
            release = Path(temporary) / "releases/0.1.0-n2.2-1234567890abcdef"
            with (
                patch.object(engine, "host_supported"),
                patch.object(
                    engine,
                    "installed",
                    return_value=({}, {"kind": "candidate", "apparmor": aa.CONTRACT}),
                ),
                patch.object(engine, "support", return_value=aa),
                patch.object(aa, "check_label", side_effect=ValueError("not attached")),
                patch.object(engine.os, "execv") as execute,
                self.assertRaisesRegex(ValueError, "not attached"),
            ):
                engine.launch(release, check_only=True)
            execute.assert_not_called()

    def test_root_helper_is_isolated_exclusive_and_has_no_global_reload(self):
        compile(aa.ROOT_HELPER, "root-policy-helper", "exec")
        for required in (
            "O_EXCL",
            "O_NOFOLLOW",
            "--skip-cache",
            "--add",
            "--remove",
            "--skip-kernel-load",
            "Existing profile preserved",
            "Changed policy preserved",
        ):
            self.assertIn(required, aa.ROOT_HELPER)
        for forbidden in (
            "--replace",
            "systemctl",
            "sysctl",
            "importlib",
            "shell=True",
        ):
            self.assertNotIn(forbidden, aa.ROOT_HELPER)
        self.assertEqual(engine.APPARMOR_CONTRACT, aa.CONTRACT)


class LauncherTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        parent = ROOT / "ci-output/native-n2-apparmor-tests-1"
        parent.mkdir(mode=0o700, exist_ok=True)
        cls.temporary = tempfile.TemporaryDirectory(dir=parent)
        cls.base = Path(cls.temporary.name)
        cls.binary = cls.base / "compiled"
        cls.build = aa.compile_bootstrap(
            ROOT / "packaging/native/bootstrap.c", cls.binary
        )
        harness = cls.base / "harness.c"
        harness.write_text(
            '#define CYRNAVA_UNIT_TEST\n#include "'
            + str(ROOT / "packaging/native/bootstrap.c")
            + '"\n'
            "int main(int argc,char **argv) { if(argc!=4)return 2; return "
            '(strcmp(argv[1],"label")==0 ? label_matches(argv[2],argv[3]) : profile_valid(argv[2],(uid_t)strtoul(argv[3],NULL,10))) ? 0:1; }\n'
        )
        cls.harness = cls.base / "harness"
        subprocess.run(
            ["/usr/bin/cc", "-std=c11", "-O2", str(harness), "-o", str(cls.harness)],
            check=True,
            capture_output=True,
            timeout=30,
        )

    @classmethod
    def tearDownClass(cls):
        cls.temporary.cleanup()

    def setUp(self):
        self.case = tempfile.TemporaryDirectory(dir=self.base)
        self.addCleanup(self.case.cleanup)
        self.root = Path(self.case.name)
        self.binary_path = self.root / "cyrnava-native"
        shutil.copyfile(self.binary, self.binary_path)
        self.binary_path.chmod(0o700)
        (self.root / "apparmor-profile").write_text(aa.profile_name(self.root) + "\n")
        (self.root / "apparmor-profile").chmod(0o600)

    def run_bootstrap(self, *args, **kwargs):
        return subprocess.run(
            [str(self.binary_path), *args],
            check=False,
            capture_output=True,
            text=True,
            timeout=5,
            **kwargs,
        )

    def test_actual_compiled_launcher_refuses_absent_attachment_without_python(self):
        result = self.run_bootstrap("--check")
        self.assertEqual(result.returncode, 1)
        self.assertIn("apparmor-profile-not-attached", result.stderr)
        self.assertNotIn(str(self.root), result.stderr)
        self.assertEqual(result.stdout, "")

    def test_arbitrary_command_arguments_are_rejected(self):
        result = self.run_bootstrap("-c", "raise RuntimeError('must not execute')")
        self.assertIn("unsupported-arguments", result.stderr)

    def test_sandbox_or_loader_environment_is_rejected(self):
        for name in (
            "QTWEBENGINE_DISABLE_SANDBOX",
            "QTWEBENGINE_CHROMIUM_FLAGS",
            "PYTHONPATH",
        ):
            with self.subTest(name=name):
                result = self.run_bootstrap(env=dict(os.environ, **{name: "synthetic"}))
                self.assertIn("unsafe-environment-override", result.stderr)

    def test_metadata_symlink_wrong_user_name_and_public_directory_refused(self):
        metadata = self.root / "apparmor-profile"
        metadata.write_text("cyrnava-native-u9999-" + "a" * 32)
        self.assertIn("profile-identity", self.run_bootstrap().stderr)
        metadata.unlink()
        metadata.symlink_to(self.binary)
        self.assertIn("profile-metadata", self.run_bootstrap().stderr)
        self.root.chmod(0o755)
        self.assertIn("private-install-root", self.run_bootstrap().stderr)

    def test_c_label_parser_does_not_accept_complain_stacking_or_prefix(self):
        name = aa.profile_name(self.root)
        for label, expected in (
            (name + " (enforce)\n", 0),
            (name + " (allow)", 0),
            (name + " (default_allow)", 0),
            (name + " (complain)", 1),
            (name + "x (enforce)", 1),
            ("outer//&" + name + " (enforce)", 1),
        ):
            with self.subTest(label=label):
                result = subprocess.run(
                    [str(self.harness), "label", label, name], check=False, timeout=5
                )
                self.assertEqual(result.returncode, expected)

    def test_c_profile_parser_rejects_wrong_user_and_nonhex_suffix(self):
        for name in (
            "cyrnava-native-u9999-" + "a" * 32,
            "cyrnava-native-u1000-" + "g" * 32,
        ):
            self.assertEqual(
                subprocess.run(
                    [str(self.harness), "profile", name, "1000"], check=False, timeout=5
                ).returncode,
                1,
            )

    def test_compiled_elf_has_hardened_headers_and_bounded_size(self):
        output = subprocess.check_output(
            ["readelf", "-h", "-l", "-d", str(self.binary)], timeout=5
        ).decode()
        self.assertIn("DYN (Position-Independent Executable file)", output)
        self.assertIn("GNU_RELRO", output)
        self.assertIn("BIND_NOW", output)
        self.assertLess(self.binary.stat().st_size, 1024**2)

    def test_real_gio_env_adapter_preserves_literal_path_and_owned_child(self):
        root = self.root / 'spaces % dollar $ and "quotes"'
        root.mkdir(mode=0o700)
        synthetic = root / "cyrnava-native"
        synthetic.write_text(
            '#!/usr/bin/python3.14\nimport json,os,sys\nprint(json.dumps({"argv":sys.argv,"pid":os.getpid(),"session":os.getsid(0)}),flush=True)\nsys.exit(7)\n'
        )
        synthetic.chmod(0o700)
        desktop = root / "synthetic.desktop"
        release = root / "releases/0.1.0-n2.2-1234567890abcdef"
        desktop.write_bytes(
            engine.desktop_entry(release, engine.CANDIDATE_ID, True, native=True)
        )
        with (root / "child.log").open("wb") as log:
            process = adapter.DesktopProcess(desktop, dict(os.environ), log)
            self.assertEqual(process.wait(5), 7)
        evidence = json.loads((root / "child.log").read_text())
        self.assertEqual(evidence["argv"], [str(synthetic)])
        self.assertEqual(evidence["pid"], process.pid)
        self.assertEqual(evidence["session"], process.pid)


class RootTransactionTests(unittest.TestCase):
    """Execute the actual root-helper logic with fake filesystem/uid/parser only."""

    def setUp(self):
        parent = ROOT / "ci-output/native-n2-apparmor-tests-1"
        parent.mkdir(mode=0o700, exist_ok=True)
        self.temporary = tempfile.TemporaryDirectory(dir=parent)
        self.addCleanup(self.temporary.cleanup)
        self.root = Path(self.temporary.name)
        self.policies = self.root / "policies"
        self.policies.mkdir(mode=0o700)
        self.profiles = self.root / "profiles"
        self.profiles.write_text("")
        self.processes = self.root / "processes"
        self.processes.mkdir()
        self.sysctl = self.root / "sysctl"
        self.sysctl.write_text("1\n")
        self.name = aa.profile_name(self.root)
        self.path = self.policies / self.name
        self.policy = b"synthetic parser fixture; never loaded"
        self.digest = __import__("hashlib").sha256(self.policy).hexdigest()
        self.code = (
            aa.ROOT_HELPER.replace("'/etc/apparmor.d'", repr(str(self.policies)))
            .replace(
                "'/sys/kernel/security/apparmor/profiles'", repr(str(self.profiles))
            )
            .replace("'/proc'", repr(str(self.processes)))
            .replace(
                "'/proc/sys/kernel/apparmor_restrict_unprivileged_userns'",
                repr(str(self.sysctl)),
            )
        )
        self.calls = []

    def run_helper(self, operation="load", *, digest=None, failure=None):
        original_stat = Path.lstat

        def stat_as_root(*args, **kwargs):
            result = list(original_stat(*args, **kwargs))
            result[4] = 0
            return os.stat_result(result)

        def parser(command, **_kwargs):
            self.calls.append(command)
            if failure and failure in command:
                raise subprocess.CalledProcessError(1, command)
            if "--add" in command:
                self.profiles.write_text(self.name + " (enforce)\n")
            if "--remove" in command:
                self.profiles.write_text("")

        with (
            patch.object(os, "geteuid", return_value=0),
            patch.object(Path, "lstat", new=stat_as_root),
            patch.object(
                sys, "argv", ["helper", operation, self.name, digest or self.digest]
            ),
            patch.object(
                sys, "stdin", type("Input", (), {"buffer": io.BytesIO(self.policy)})()
            ),
            patch.object(subprocess, "run", side_effect=parser),
            contextlib.redirect_stdout(io.StringIO()) as output,
        ):
            # Execute only the checked-in literal with fixture paths and mocked root/parser.
            exec(compile(self.code, "root-helper-fixture", "exec"), {})  # noqa: S102
        return json.loads(output.getvalue())

    def test_exact_policy_load_parses_then_adds_without_replacing(self):
        result = self.run_helper()
        self.assertEqual(result["status"], "PASS")
        self.assertEqual(self.path.read_bytes(), self.policy)
        self.assertEqual(len(self.calls), 2)
        self.assertIn("--skip-kernel-load", self.calls[0])
        self.assertIn("--add", self.calls[1])

    def test_bad_digest_does_not_create_or_load(self):
        with self.assertRaisesRegex(SystemExit, "digest refused"):
            self.run_helper(digest="0" * 64)
        self.assertFalse(self.path.exists())
        self.assertEqual(self.calls, [])

    def test_existing_file_and_loaded_name_are_preserved(self):
        self.path.write_bytes(b"owner file")
        with self.assertRaises(FileExistsError):
            self.run_helper()
        self.assertEqual(self.path.read_bytes(), b"owner file")
        self.path.unlink()
        self.profiles.write_text(self.name + " (enforce)\n")
        with self.assertRaisesRegex(SystemExit, "Existing profile"):
            self.run_helper()
        self.assertFalse(self.path.exists())
        self.assertEqual(self.calls, [])

    def test_failed_load_preserves_policy_without_automatic_retry(self):
        with self.assertRaises(subprocess.CalledProcessError):
            self.run_helper(failure="--add")
        self.assertEqual(self.path.read_bytes(), self.policy)
        self.assertEqual(len(self.calls), 2)

    def test_removal_refuses_changed_file_or_live_application(self):
        self.path.write_bytes(b"changed")
        with self.assertRaisesRegex(SystemExit, "Changed policy"):
            self.run_helper("remove")
        self.path.write_bytes(self.policy)
        self.path.chmod(0o600)
        attr = self.processes / "123/attr/apparmor"
        attr.mkdir(parents=True)
        (attr / "current").write_text(self.name + " (enforce)\n")
        with self.assertRaisesRegex(SystemExit, "Live profile"):
            self.run_helper("remove")
        self.assertTrue(self.path.exists())
        self.assertEqual(self.calls, [])

    def test_exact_unused_policy_can_be_removed_without_global_change(self):
        self.path.write_bytes(self.policy)
        self.path.chmod(0o600)
        self.profiles.write_text(self.name + " (enforce)\n")
        self.assertEqual(self.run_helper("remove")["status"], "PASS")
        self.assertFalse(self.path.exists())
        self.assertEqual(len(self.calls), 1)
        self.assertIn("--remove", self.calls[0])
        self.assertEqual(self.sysctl.read_text(), "1\n")

    def test_changed_global_restriction_refuses_without_writing(self):
        self.sysctl.write_text("0\n")
        with self.assertRaisesRegex(SystemExit, "Global namespace restriction differs"):
            self.run_helper()
        self.assertFalse(self.path.exists())
        self.assertEqual(self.calls, [])


class TransactionTests(install_tests.InstallTests):
    """Reuse all archive/trust/transaction cases with the new policy contract."""

    def make_archive(self, extra=None, mode=None):
        extra = dict(extra or {})
        extra.update(
            {
                "cyrnava-native": b"synthetic-not-executed",
                "bootstrap.c": b"synthetic fixture source",
                "apparmor_support.py": (
                    ROOT / "packaging/native/apparmor_support.py"
                ).read_bytes(),
            }
        )
        super().make_archive(extra=extra, mode=mode)
        self.value["apparmor"] = aa.CONTRACT
        self.value["files"]["cyrnava-native"]["mode"] = 0o700
        self.save_manifest()

    def test_archive_exact_inventory_extracts(self):
        output = self.base / "extracted"
        output.mkdir()
        engine.extract(self.archive, output, engine.validate_manifest(self.value))
        self.assertEqual(
            len([p for p in output.rglob("*") if p.is_file()]), len(self.value["files"])
        )

    def test_desktop_command_accepts_normal_venv_interpreter_link(self):
        with self.fixtures():
            result = self.install()
        release = Path(result["release"])
        text = engine.desktop_entry(release, engine.CANDIDATE_ID, True, native=True)
        self.assertIn(b"Exec=/usr/bin/env -- ", text)
        self.assertIn(b"cyrnava-native", text)
        self.assertNotIn(b".venv/bin/python", text)

    def test_incomplete_policy_inventory_and_nonexecutable_bootstrap_refused(self):
        del self.value["files"]["bootstrap.c"]
        self.value["expanded_bytes"] = sum(
            f["bytes"] for f in self.value["files"].values()
        )
        with self.assertRaisesRegex(ValueError, "AppArmor component"):
            engine.validate_manifest(self.value)
        self.make_archive()
        self.value["files"]["cyrnava-native"]["mode"] = 0o600
        with self.assertRaisesRegex(ValueError, "must be executable"):
            engine.validate_manifest(self.value)

    def test_changed_native_bootstrap_is_preserved_during_recovery(self):
        original = engine.journal_checkpoint

        def interrupt(root, journal, phase):
            original(root, journal, phase)
            if phase == "activated":
                raise KeyboardInterrupt

        with (
            self.fixtures(),
            patch.object(engine, "journal_checkpoint", side_effect=interrupt),
            self.assertRaises(KeyboardInterrupt),
        ):
            self.install()
        (self.root / "cyrnava-native").write_bytes(b"changed")
        with self.assertRaisesRegex(ValueError, "Changed transaction"):
            engine.recover(self.root, disposable=True)
        self.assertEqual((self.root / "cyrnava-native").read_bytes(), b"changed")
